import assert from 'node:assert/strict';
import { readFile, writeFile } from 'node:fs/promises';
import { createHash } from 'node:crypto';
const root = 'docs/tasks/audit-contracts/natural-preparation/20261029';
const report = JSON.parse(await readFile('test-results/pr109/buildings-conservative/natural-play-report.json', 'utf8'));
const hash = data => createHash('sha256').update(data).digest('hex');
assert.equal(report.status, 'passed'); assert.equal(report.minutes, 10); assert.equal(report.world, 'buildings');
const duration = (Date.parse(report.finishedAt) - Date.parse(report.startedAt)) / 1000;
assert.ok(duration >= 600); assert.deepEqual(report.errors, []); assert.deepEqual(report.externalRequests, []);
assert.equal(report.preparationBlocked, undefined);
assert.equal(report.htmlSha256, hash(await readFile('dist/index.html')));
const previous = new Map();
for (const sample of report.samples) {
  assert.ok(Number.isInteger(sample.kills) && sample.kills >= 0);
  assert.ok(sample.kills >= (previous.get(sample.run) ?? 0), 'cumulative kills cannot decrease within a run');
  assert.ok(sample.kills <= report.outcomes[sample.run - 1].kills, 'samples must not exceed their committed settlement');
  previous.set(sample.run, sample.kills);
}
const totalKills = report.outcomes.reduce((n, r) => n + r.kills, 0);
assert.equal(totalKills, report.finalStats.kills, 'all committed run results must reconcile with final profile statistics');
assert.ok(report.containerLoot.opened.length > 0); assert.ok(report.containerLoot.transfers.length > 0);
assert.ok(Object.keys(report.enemyPhases).some(key => key.endsWith('/windup')));
const sourceFiles = ['src/raid-runtime/test-support.ts', 'tests/raid-runtime.test.ts', 'scripts/natural-play.mjs', 'scripts/natural-play-state.mjs', 'scripts/natural-play-preparation.mjs', 'scripts/natural-play-policy.mjs', 'scripts/ci/natural-play-state.test.mjs', 'scripts/ci/natural-play-policy.test.mjs'];
const sourceHashes = Object.fromEntries(await Promise.all(sourceFiles.map(async path => [path, hash(await readFile(path))])));
const summary = { status: 'passed', baseline: '275fafd77b1574938ecc0409ab405b77850359c1', sourceHashes,
  artifacts: { htmlBytes: 9260294, portableZipBytes: 6035876, webZipBytes: 6031367 },
  environment: { platform: process.platform, node: process.version, pnpm: '11.25.0', browser: report.browser },
  checks: { unit: '1058/1058', guards: '123/123', package: 'passed', medicalOracle: '756/756 matched, no input mutation', legacySamplerControl: '3 passed / 1 expected failure: event eviction produces 0 instead of 1' },
  natural: { startedAt: report.startedAt, finishedAt: report.finishedAt, durationSeconds: duration, htmlSha256: report.htmlSha256, timing: report.timing,
    outcomes: report.outcomes.map(r => ({ outcome: r.outcome, kills: r.kills })), totalKills, samples: report.samples.length, sampledRuns: previous.size,
    monotonicWithinEveryRun: true, samplesNotAboveSettlement: true, finalStatsReconciled: true, opened: report.containerLoot.opened.length, nativeTransfers: report.containerLoot.transfers.length,
    banks: report.preparation.filter(d => d.kind === 'bank-loot').length, sales: report.preparation.filter(d => d.kind === 'sell-loot').length,
    enemyWindups: Object.keys(report.enemyPhases).filter(key => key.endsWith('/windup')), pageErrors: report.errors.length, externalRequests: report.externalRequests.length },
  retained: { firstFailure: 'insufficient-funds-report.json', secondFailure: 'unbanked-loot-report.json', thirdFailure: 'repeated-deaths-report.json', interruptedRun: 'interrupted-progress.json (no final report; not counted as pass)' },
  boundaries: 'Automated natural input, not human balance or physical-device acceptance. UI/quota/funding and kill-control are separate explicit fixtures. Save format, game rules, ordinary HUD, #32 F20 unchanged.' };
await writeFile(root + '/verification.json', JSON.stringify(summary, null, 2) + '\n');
console.log(JSON.stringify(summary.natural, null, 2));
