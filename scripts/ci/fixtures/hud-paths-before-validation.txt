// Auxiliary strong root-path inspection; excludes weak snapshot edges and omits string content.
import{readFile,writeFile}from'node:fs/promises';const [out,...files]=process.argv.slice(2);if(!out||!files.length)throw Error('output snapshot...');const rows=[];for(const file of files){const s=JSON.parse(await readFile(file,'utf8')),f=s.snapshot.meta.node_fields,n=f.length,ef=s.snapshot.meta.edge_fields,en=ef.length,T=f.indexOf('type'),N=f.indexOf('name'),ID=f.indexOf('id'),EC=f.indexOf('edge_count'),ET=ef.indexOf('type'),EN=ef.indexOf('name_or_index'),TO=ef.indexOf('to_node'),types=s.snapshot.meta.node_types[0],ets=s.snapshot.meta.edge_types[0],count=s.nodes.length/n,starts=new Uint32Array(count+1);for(let i=0,e=0;i<count;i++){starts[i]=e;e+=s.nodes[i*n+EC]*en;}starts[count]=s.edges.length;const desc=i=>({id:s.nodes[i*n+ID],type:types[s.nodes[i*n+T]],name:types[s.nodes[i*n+T]].includes('string')?'(omitted)':s.strings[s.nodes[i*n+N]].slice(0,250)}),parent=new Int32Array(count).fill(-1),via=new Uint32Array(count),q=new Uint32Array(count);let tail=1;parent[0]=0;for(let h=0;h<tail;h++){const i=q[h];for(let e=starts[i];e<starts[i+1];e+=en){if(ets[s.edges[e+ET]]==='weak')continue;const to=s.edges[e+TO]/n;if(parent[to]<0){parent[to]=i;via[to]=e;q[tail++]=to;}}}const hud=[];for(let i=0;i<count;i++)if(desc(i).type==='native'&&/cs-hud/.test(desc(i).name)){const path=[];if(parent[i]>=0){for(let j=i,steps=0;steps<80;steps++){const e=via[j],type=ets[s.edges[e+ET]],v=s.edges[e+EN];path.push({...desc(j),...(j?{via:type+':'+(['element','hidden'].includes(type)?v:s.strings[v])}:{})});if(!j)break;j=parent[j];}path.reverse();}hud.push({node:desc(i),reachable:parent[i]>=0,path});}rows.push({file,hud});}await writeFile(out,JSON.stringify(rows,null,2)+'\n');console.log(JSON.stringify(rows));
